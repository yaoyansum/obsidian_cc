# 5. 构建核平滑连续面

def build_kernel_surface(

    data,

    lon_col,

    lat_col,

    slope_col,

    mode="signed",

    resolution=0.25,

    sigma_degree=3.0,

    clip_percentile=98,

    support_threshold=0.005,

    lat_min=-60,

    lat_max=85,

):

    """

    mode:

        "increase" : 仅使用正 slope，权重 = slope

        "decrease" : 仅使用负 slope，权重 = abs(slope)

        "signed"   : 使用全部 slope，正负一起

    """

  

    d = data[[lon_col, lat_col, slope_col]].dropna().copy()

  

    if mode == "increase":

        d = d[d[slope_col] > 0].copy()

        d["weight"] = d[slope_col]

  

    elif mode == "decrease":

        d = d[d[slope_col] < 0].copy()

        d["weight"] = np.abs(d[slope_col])

  

    elif mode == "signed":

        d = d[d[slope_col] != 0].copy()

        d["weight"] = d[slope_col]

  

    else:

        raise ValueError("mode 必须是 increase / decrease / signed")

  

    if len(d) == 0:

        raise ValueError(f"{mode} 模式下没有可用数据，请检查 slope 或 p 值筛选条件。")

  

    # 截断极端值

    if mode in ["increase", "decrease"]:

        upper = np.nanpercentile(d["weight"], clip_percentile)

        d["weight"] = d["weight"].clip(upper=upper)

    else:

        upper = np.nanpercentile(np.abs(d["weight"]), clip_percentile)

        d["weight"] = d["weight"].clip(lower=-upper, upper=upper)

  

    lon_edges = np.arange(-180, 180 + resolution, resolution)

    lat_edges = np.arange(lat_min, lat_max + resolution, resolution)

  

    lon = d[lon_col].to_numpy()

    lat = d[lat_col].to_numpy()

    weight = d["weight"].to_numpy()

  

    # 加权和

    weighted_sum, _, _ = np.histogram2d(

        lat, lon,

        bins=[lat_edges, lon_edges],

        weights=weight

    )

  

    # 数量密度

    count, _, _ = np.histogram2d(

        lat, lon,

        bins=[lat_edges, lon_edges]

    )

  

    sigma_pixel = sigma_degree / resolution

  

    smooth_weighted = gaussian_filter(

        weighted_sum,

        sigma=sigma_pixel,

        mode="constant",

        cval=0

    )

  

    smooth_count = gaussian_filter(

        count,

        sigma=sigma_pixel,

        mode="constant",

        cval=0

    )

  

    with np.errstate(divide="ignore", invalid="ignore"):

        surface = smooth_weighted / smooth_count

  

    # 支撑强度归一化

    if np.nanmax(smooth_count) > 0:

        support = smooth_count / np.nanmax(smooth_count)

    else:

        support = np.zeros_like(smooth_count)

  

    # 无湖区掩膜

    surface[support < support_threshold] = np.nan

  

    lon_centers = (lon_edges[:-1] + lon_edges[1:]) / 2

    lat_centers = (lat_edges[:-1] + lat_edges[1:]) / 2

  

    return lon_centers, lat_centers, surface, support

# 6. 绘图函数：双向等值线图

# =========================================================

def plot_signed_contour_map(

    lon,

    lat,

    Z,

    output_png,

    title="Contour hotspots of lake DOC storage change",

    cbar_label="Kernel-smoothed DOC storage trend"

):

    projection = ccrs.Robinson()

    data_crs = ccrs.PlateCarree()

  

    fig = plt.figure(figsize=(13.5, 7.0))

    ax = plt.axes(projection=projection)

    ax.set_global()

  

    # 底图

    ax.add_feature(cfeature.OCEAN, facecolor="#eef5fb", zorder=0)

    ax.add_feature(cfeature.LAND, facecolor="#fbfbf7", edgecolor="none", zorder=0)

    ax.add_feature(cfeature.COASTLINE, linewidth=0.45, edgecolor="0.35", zorder=3)

    ax.add_feature(cfeature.BORDERS, linewidth=0.25, edgecolor="0.55", zorder=3)

  

    # 对称色带

    abs_max = np.nanpercentile(np.abs(Z), 98)

    if not np.isfinite(abs_max) or abs_max == 0:

        abs_max = np.nanmax(np.abs(Z))

    if not np.isfinite(abs_max) or abs_max == 0:

        abs_max = 1

  

    levels = np.linspace(-abs_max, abs_max, n_levels_signed)

  

    # 填色等值面

    cf = ax.contourf(

        lon,

        lat,

        Z,

        levels=levels,

        transform=data_crs,

        cmap="RdBu_r",

        extend="both",

        zorder=1

    )

  

    # 普通等值线

    cs = ax.contour(

        lon,

        lat,

        Z,

        levels=levels,

        colors="k",

        linewidths=0.40,

        alpha=0.55,

        transform=data_crs,

        zorder=2

    )

  

    # 强调 0 等值线

    try:

        cs0 = ax.contour(

            lon,

            lat,

            Z,

            levels=[0],

            colors="k",

            linewidths=1.0,

            transform=data_crs,

            zorder=3

        )

    except:

        pass

  

    try:

        ax.clabel(cs, inline=True, fontsize=7, fmt="%.2f")

    except:

        pass

  

    ax.gridlines(

        draw_labels=False,

        linewidth=0.25,

        color="gray",

        alpha=0.35,

        linestyle="--"

    )

  

    ax.set_title(title, fontsize=14, pad=12)

  

    cbar = plt.colorbar(

        cf,

        ax=ax,

        orientation="horizontal",

        pad=0.045,

        shrink=0.72,

        aspect=35

    )

    cbar.set_label(cbar_label, fontsize=11)

    cbar.ax.tick_params(labelsize=9)

  

    plt.tight_layout()

    plt.savefig(output_png, dpi=dpi, bbox_inches="tight")

    plt.close()

  

    print(f"已保存: {output_png}")
