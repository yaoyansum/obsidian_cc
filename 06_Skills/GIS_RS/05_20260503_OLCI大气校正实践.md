### GEE 辅助数据获取

#### AOT (气溶胶光学厚度)

从 MODIS MCD19A2 获取 550nm 气溶胶光学厚度：

```python
import ee

def aot_gee(TopLeftLon, BottomRightLat, TopRightLon, TopRightLat, Year, Moon, Day):
    ee.Initialize()
    
    roi = ee.Geometry.Rectangle([TopLeftLon, BottomRightLat, TopRightLon, TopRightLat])
    start_date = ee.Date(f"{Year}-{Moon}-{Day}")
    end_date = start_date.advance(1, "day")
    
    dataset = (ee.ImageCollection('MODIS/061/MCD19A2_GRANULES')
               .select('Optical_Depth_055')
               .filterDate(start_date, end_date)
               .filterBounds(roi))
    
    mean_aot = dataset.mean().clip(roi)
    mean_aot_value = mean_aot.reduceRegion(
        reducer=ee.Reducer.mean(),
        geometry=roi,
        scale=500,
        maxPixels=1e9
    )
    
    aot = mean_aot_value.get('Optical_Depth_055').getInfo()
    aot = aot / 1000
    print('Mean AOT (550nm) in ROI:', aot)
    return aot
```

def meanDEM_gee(minLon, minLat, maxLon, maxLat):
    ee.Initialize()
    roi = ee.Geometry.Rectangle([minLon, minLat, maxLon, maxLat])
    SRTM = ee.Image('CGIAR/SRTM90_V4')
    alt = SRTM.reduceRegion(
        reducer=ee.Reducer.mean(),
        geometry=roi,
        scale=500,
        maxPixels=1e9).get('elevation').getInfo()
    meanDEM = alt / 1000
    print("meanDEM:", meanDEM)
    return meanDEM

### 6S 大气校正函数

```python
from Py6S import *

def AtmosphericCorrection(BandId, solar_z, solar_a, view_z, view_a,
                          Day, Month, ImageCenterLat, aot, meanDEM):
    """
    input: BandId, solar_z, solar_a, view_z, view_a,
           Day, Month, h2o, o3, aot, meanDEM
    output: (xa, xb, xc)
    """
    s = SixS()
    s.geometry = Geometry.User()
    
    s.geometry.solar_z = solar_z
    s.geometry.solar_a = solar_a
    s.geometry.view_z = view_z
    s.geometry.view_a = view_a
    s.geometry.month = int(Month)
    s.geometry.day = int(Day)
    
    # 大气模式类型（根据纬度自动选择）
    if ImageCenterLat > -15 and ImageCenterLat < 15:
        s.atmos_profile = AtmosProfile.PredefinedType(AtmosProfile.Tropical)
    if ImageCenterLat > 15 and ImageCenterLat < 45:
        if s.geometry.month > 4 and s.geometry.month < 9:
            s.atmos_profile = AtmosProfile.PredefinedType(AtmosProfile.MidlatitudeSummer)
        else:
            s.atmos_profile = AtmosProfile.PredefinedType(AtmosProfile.MidlatitudeWinter)
    if ImageCenterLat > 45 and ImageCenterLat < 60:
        if s.geometry.month > 4 and s.geometry.month < 9:
            s.atmos_profile = AtmosProfile.PredefinedType(AtmosProfile.SubarcticSummer)
        else:
            s.atmos_profile = AtmosProfile.PredefinedType(AtmosProfile.SubarcticWinter)
    
    # 气溶胶类型：大陆型
    s.aero_profile = AtmosProfile.PredefinedType(AeroProfile.Continental)
    
    # 下垫面反射率（开阔海洋约0.02-0.05）
    s.ground_reflectance = GroundReflectance.HomogeneousLambertian(0.36)
    
    # 550nm气溶胶光学厚度
    s.aot550 = aot
    
    # 海拔高度设置
    s.altitudes = Altitudes()
    s.altitudes.set_target_custom_altitude(meanDEM)
    s.altitudes.set_sensor_satellite_level()
    
    # OLCI波段设置
    band_map = {
        1: PredefinedWavelengths.S3A_OLCI_01,
        2: PredefinedWavelengths.S3A_OLCI_02,
        # ... (3-20)
        21: PredefinedWavelengths.S3A_OLCI_21,
    }
    
    if BandId in band_map:
        s.wavelength = Wavelength(band_map[BandId])
    
    # 大气校正模式
    s.atmos_corr = AtmosCorr.AtmosCorrLambertianFromReflectance(-0.1)
    
    # 运行6S
    s.run()
    xa = s.outputs.coef_xa
    xb = s.outputs.coef_xb
    xc = s.outputs.coef_xc
    
    return (xa, xb, xc)
```

### ROI 信息提取

从校正后的 NetCDF 文件中提取指定坐标点的波段信息：

```python
def roi_information(directory_path, excel_file):
    """
    从校正结果中提取坐标点的波段信息
    input: (lon,lat) 坐标文件路径, 校正结果目录
    output: CSV文件路径
    """
    output_csv_file = os.path.join(directory_path, "坐标点with_bands.csv")
    
    roi_df = pd.read_excel(excel_file)
    properties_df = pd.DataFrame()
    
    entries = os.listdir(directory_path)
    files_geo = [f for f in entries if "geo" in f and "tie" not in f
                 and os.path.isfile(os.path.join(directory_path, f))]
    geo_nc = os.path.join(directory_path, files_geo[0])
    
    radiance_files = [f for f in entries if "radiance.nc" in f
                      and os.path.isfile(os.path.join(directory_path, f))]
    
    ds = xr.open_dataset(geo_nc)
    lon = ds['longitude'].values
    lat = ds['latitude'].values
    
    for index, row in roi_df.iterrows():
        lat_point, lon_point = row['Latitude'], row['Longitude']
        dist = np.sqrt((lat - lat_point) ** 2 + (lon - lon_point) ** 2)
        row_idx, col_idx = np.unravel_index(np.argmin(dist), dist.shape)
        
        result = []
        for file in radiance_files:
            in_file = os.path.join(directory_path, file)
            data = xr.open_dataset(in_file)
            values = data[file[:-3]].values[row_idx, col_idx]
            result.append(values)
        
        properties_df.loc[index, 'Latitude'] = lat_point
        properties_df.loc[index, 'Longitude'] = lon_point
        for i, value in enumerate(result):
            properties_df.loc[index, f'Band_{i+1}'] = value
    
    properties_df.to_csv(output_csv_file, index=False)
    return output_csv_file
```

def block(full_path):
    """
    对OLCI影像进行批处理大气校正
    """
    entries = os.listdir(full_path)
    
    # 读取坐标文件
    files_geo = [f for f in entries if "geo" in f and "tie" not in f
                 and os.path.isfile(os.path.join(full_path, f))]
    geo_nc = os.path.join(full_path, files_geo[0])
    
    ds = xr.open_dataset(geo_nc)
    lon = ds['longitude'].values
    lat = ds['latitude'].values
    
    # 计算中心点坐标和时空参数
    lon_mean = (lon.max() + lon.min()) / 2
    lat_mean = (lat.max() + lat.min()) / 2
    
    attrs_dic = ds.attrs
    datetime_str = attrs_dic['start_time']
    Year = datetime_str[0:4]
    Month = datetime_str[5:7]
    Day = datetime_str[8:10]
    
    # 获取辅助数据
    meanDEM = meanDEM_gee(lon.min(), lat.min(), lon.max(), lat.max())
    aot = aot_gee(lon.min(), lat.min(), lon.max(), lat.max(), Year, Month, Day)
    
    # 获取观测几何
    solar_z, solar_a, view_z, view_a = solar_view(full_path)
    
    # 计算各波段校正系数
    df = pd.DataFrame()
    for i in range(1, 22):
        xa, xb, xc = AtmosphericCorrection(
            i, solar_z, solar_a, view_z, view_a,
            Day, Month, lat_mean, aot, meanDEM
        )
        df["Band" + str(i)] = [xa, xb, xc]
    
    output_csv_file = os.path.join(full_path, "矫正方程with_bands.csv")
    df.to_csv(output_csv_file, index=False)
    return output_csv_file
